וִדּוּי דְּבָרִים

עַל פִּי וִדּוּיוֹ שֶׁל הֶחָסִיד רַבִּי יִצְחָק בְּרַיְיטֶער זַצַ"ל הי"ד

אָנָּא ה' אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ. תָּבֹא לְפָנֶיךָ תְּפִלָּתֵנוּ, וְאַל תִּתְעַלַּם מַלְכֵּנוּ מִתְּחִנָּתֵנוּ. שֶׁאֵין אֲנַחְנוּ עַזֵּי פָנִים וּקְשֵׁה עֹרֶף לוֹמַר לְפָנֶיךָ ה' אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ צַדִּיקִים אֲנַחְנוּ וְלֹא חָטָאנוּ. אֲבָל חָטָאנוּ. עָוִינוּ. פָּשַׁעְנוּ. אֲנַחְנוּ וַאֲבוֹתֵינוּ וְאַנְשֵׁי בֵיתֵנוּ:

אָשַׁמְנוּ. בָּגַדְנוּ. גָּזַלְנוּ. דִּבַּרְנוּ דֹפִי וְלָשׁוֹן הָרָע. הֶעֱוִינוּ. וְהִרְשַׁעְנוּ. זַדְנוּ. חָמַסְנוּ. טָפַלְנוּ שֶׁקֶר וּמִרְמָה. יָעַצְנוּ עֵצוֹת רָעוֹת. כִּזַּבְנוּ. כָּעַסְנוּ. לַצְנוּ. מָרַדְנוּ. מָרִינוּ דְבָרֶיךָ. נִאַצְנוּ. נִאַפְנוּ. סָרַרְנוּ. עָוִינוּ. פָּשַׁעְנוּ. פָּגַמְנוּ. צָרַרְנוּ. צִעַרְנוּ אָב וָאֵם. קִשִּׁינוּ עֹרֶף. רָשַׁעְנוּ. שִׁחַתְנוּ. תִּעַבְנוּ. תָּעִינוּ וְתִעֲתַעְנוּ. וְסַרְנוּ מִמִּצְווֹתֶיךָ וּמִמִּשְׁפָּטֶיךָ הַטּוֹבִים וְלֹא שָׁוָה לָנוּ. וְאַתָּה צַדִּיק עַל כָּל הַבָּא עָלֵינוּ. כִּי אֱמֶת עָשִׂיתָ וַאֲנַחְנוּ הִרְשָׁעְנוּ:

אֶל כְּבוֹד נִשְׁמַת אֲדוֹנֵינוּ מוֹרֵינוּ וְרַבֵּנוּ, אוֹר הָאוֹרוֹת, צַח הָצַחְצַחוֹת, סַבָּא דְסַבִּין, עַתִּיק דְעַתִּיקִין, הַנָּהָר הַיּוֹצֵא מֵעֵדֶן, לְהַשְׁקוֹת, לְהַחֲיוֹת וּלְטַהֵר כָּל הַבְּרוּאִים וְכָל הַנְּשָׁמוֹת, הַלֹּא הוּא רַבֵּנוּ הַקָּדוֹשׁ רַבִּי נַחְמָן בֶּן פֵיגֶא, זְכוּתוֹ יָגֵן עָלֵינוּ וְעַל כָּל יִשְׂרָאֵל, אָמֵן. בִּדְחִילוּ וּרְחִימוּ הֲרֵינִי נִגַּשׁ אֶל מְקוֹם גְּנִיזַתְכֶם, מְקוֹם קֹדֶשׁ הַקָּדָשִׁים, מְקוֹם נְבִיעַת הַמַּעְיָן הַקָּדוֹשׁ, מִשָּׁם כָּל הָעֲדָרִים שׁוֹתִים וְאֵלָיו כָּל הַלְּבָבוֹת נִכְסָפִים, לְקַיֵּם צִוּוּיוֹ בְּסִפְרוֹ הַקָּדוֹשׁ לִקּוּטֵי מוֹהֲרַ"ן בְּתוֹרָה ד', וְלַעֲשׂוֹת וִדּוּי דְּבָרִים לִפְנֵי הַצַּדִּיק, כִּי אֵינִי יוֹדֵעַ בְּאֵיזֶה דֶּרֶךְ אֲנִי מְהַלֵּךְ, וְאַתָּה אָמַרְתָּ כִּי בָּאתָ לָעוֹלָם לְקָרֵב נְפָשׁוֹת יִשְׂרָאֵל אֶל הַשֵּׁם יִתְבָּרַךְ, אֲבָל אֵינְךָ יָכֹל לְקָרֵב כִּי אִם מִי שֶׁבָּא אֶצְלְךָ וּמְסָפֵּר לְךָ חֶסְרוֹנוֹ אַתָּה יָכוֹל לְתַקְּנוֹ. וְגַם אָמַרְתָּ בְּלִקּוּטֵי מוֹהֲרַ"ן תּוֹרָה קפ"ח, שֶׁצָּרִיךְ לִנְסֹעַ לַצַּדִּיק לְהַחְזִיר אֲבֵדָתוֹ, כִּי כָּל הָאֲבֵדוֹת הֵן אֵצֶל הַצַּדִּיק.

רַבִּי, רַבִּי, אֵינִי יוֹדֵעַ כְּלָל מַה אִבַּדְתִּי, לֹא כָּל שֶׁכֵּן שֶׁאֵינִי יוֹדֵעַ לְהַכִּיר אֲבֵדוֹתַי, אַךְ אֲנִי מַאֲמִין בֶּאֱמוּנָה שְׁלֵמָה, שֶׁעֶצֶם טִרְחָתִי וּבִיאָתִי לְכָאן, וְדִבּוּרַי אֵלֶּה, לַמְרוֹת רֹב הַבִּלְבּוּל וְהַפְּנִיּוֹת שֶׁבָּהֶם, יוֹעִיל לִי שֶׁתַּעֲסֹק בְּתִּקּוּן נִשְׁמָתִי, וְתָשִׁיב לִי אֶת אֲבֵדוֹתַי הָרַבּוֹת שֶׁאִבַּדְתִּי, הֵן בְּגִּלְגּוּל זֶה וְהֵן בְּגִּלְגּוּלִים אֲחֵרִים.

(כָּאן יֵשׁ לְהִשְׁתַּדֵּל לְסַפֵּר אֶת כָּל הַמַּעֲשִׂים, הַדִּבּוּרִים, וְהַמַּחְשָׁבוֹת הָרָעוֹת בָּהֶם נִכְשַׁלְנוּ מִיּוֹם עָמְדֵּנוּ עַל דַּעְתֵּנוּ עַד הַיּוֹם הַזֶּה).

רַבִּי, רַבִּי, מַר וְצַר לִי מְאוֹד, חֹשֶׁךְ עָנָן וַעֲרָפֶל. וּמַה אֶעֱשֶׂה? וּמִי יָשׁוּב בַּעֲדִי? הַלֹּא עַל זֶה נִבְרֵאתִי, שֶׁאָשׁוּב בְּעַצְמִי וְאֶתְגַּבֵּר עַל כֹּחוֹת הָרַע וְאֶסְבֹּל צַעַר מִכְּבִישַׁת הַתַּעֲנוּג שֶׁל הַיֵּצֶר. וְכָל הַצַּדִּיקִים וְהַיְּרֵאִים עָבְרוּ מַעֲבָר זֶה, כִּי גַּם אוֹתָם עִנָּה הַיֵּצֶר הָרַע וְרָצָה לִתֵּן לָהֶם הַתַּעֲנוּג הֶחָשׁוּךְ וְהַמַּר הַזֶּה. אֲבָל הֵם זָכוּ לְהִתְגַּבֵּר וּלְהַעֲבִיר עֵינָם וְלִבָּם מֵהָאי עַלְמָא (מִתַּעְתּוּעֵי הָעוֹלָם הַזֶּה) וְדָבְקוּ אֶת עַצְמָם בְּהַשֵּׁם יִתְבָּרַךְ וְזָכוּ לְמַה שֶׁזָּכוּ, אַשְׁרֵי לָהֶם. אֲבָל אֲנִי אוֹי לִי, כִּי אֲנִי רוֹצֶה לַעֲבֹד אֶת ה' עִם כָּל הַתַּאֲווֹת. הַלֹּא עַל מִי תְּלוּנוֹתַי, כִּי אִם עַל נַפְשִׁי, אֲבָל עַל זֶה בְּעַצְמוֹ אֲנִי צוֹעֵק וּבָא לִפְנֵי קְדֻשַּׁתְכֶם, שֶׁתַּעֲשׂוּ עִמִּי דָּבָר כָּזֶה, שֶׁגַּם אֲנִי אַשִּׂיג כֹּחַ לַעֲשׂוֹת כְּמוֹ כָּל הַצַּדִּיקִים אֲשֶׁר מֵעוֹלָם, שֶׁאֶזְכֶּה מֵעַתָּה לִחְיוֹת חַיִּים טוֹבִים בִּקְּדֻשָּׁה וּבְטָּהֳרָה, וּלְבַלּוֹת יָמַי עַל הַתּוֹרָה וְעַל הָעֲבוֹדָה, וַאֲנִי מוֹסֵר מוֹדָעָה לִפְנֵיכֶם, לִפְנֵי שֹׁרֶשׁ נְשָׁמוֹת יִשְׂרָאֵל, צַדִּיק מוֹשֵׁל בְּיִרְאַת אֱלֹקִים, כִּי אֲנִי רוֹצֶה לַעֲבֹד אֶת הַשֵּׁם יִתְבָּרָךְ מֵעַתָּה וְעַד עוֹלָם וְלִכְבֹּשׁ אֶת יִצְרִי וְלִחְיוֹת חַיִּים אֲמִתִּיִּים וּקְדוֹשִׁים כִּרְצוֹן ה' בֶּאֱמֶת מֵעַתָּה וְעַד עוֹלָם. וְאַתָּה עֲשֵׂה עִמִּי בְּכֹחֲךָ הַגָּדוֹל וְתוֹצִיא אֶת נִשְׁמָתִי מִכָּל מִינֵי גֵּיהֵנֹּם הַשּׁוֹלְטִים בָּהּ, לְאוֹר בְּאוֹר הַחַיִּים וּלְבַקֵּר בְּהֵיכָלוֹ. וְאֶזְכֶּה לְדֶרֶךְ יְשָׁרָה לָלֶכֶת בָּהּ לְפִי שֹׁרֶשׁ נִשְׁמָתִי וּמַהוּתִי, בְּאֹפֶן שֶׁגַּם עַל־יָדִי יִתְקַדֵּשׁ שְׁמוֹ יִתְבָּרַךְ בָּעוֹלָם הַזֶּה וּבְכָל הָעוֹלָמוֹת כֻּלָּם. אָמֵן.

חָזָק חָזָק וְנִתְחַזַּק: תָּם וְנִשְׁלָם שֶׁבַח לָאֵל בּוֹרֵא עוֹלָם. זְכוּת רַבֵּנוּ הַקָּדוֹשׁ יָגֵן עָלֵינוּ וְעַל כָּל אֶחָד מִיִּשְׂרָאֵל, זְכוּת נַ נַחְ נַחְמָ נַחְמָן מֵאוּמָן.
