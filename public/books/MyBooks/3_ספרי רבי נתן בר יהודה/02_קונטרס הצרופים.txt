@ הַקְדָּמַת הַמְחַבֵּר
אֲנִי הֶעָנִי וְהָאֶבְיוֹן, הֶחָסֵר מִשֵּׂכֶל זַךְ וָצַח, וּמִן רַעֲיוֹנִים זַכִּים וּטְהוֹרִים – לְהַשִּׂיג בְּזִכְרוֹן בְּמֹחִי כָּל דִּבּוּר וְדִבּוּר מַאֲמָרוֹת טְהוֹרוֹת כֶּסֶף צָרוּף מְזֻקָּק שִׁבְעָתָיִם, מֵעֲרֵבוּת נְעִימוּת מְתִיקַת יְדִידוּת מִסִּפְרֵי רַבֵּנוּ הַנַּחַל נֹבֵעַ מְקוֹר חָכְמָה – אֲשֶׁר נוֹעַם שְׁמוֹ הִמְתִּיק בְּזֶה הַפָּסוּק, וְאוֹר תּוֹרָתוֹ מֵאִיר בְּכָל הָעוֹלָמוֹת, וְהוּא הִתְחִיל בַּתִּקּוּן הַכְּלָלִי וְיִגְמֹר. 
וַאֲנִי שָׁמַעְתִּי מִתַּלְמִידוֹ הַקָּדוֹשׁ אֲשֶׁר סָמַךְ יָדָיו גְּלִילֵי זָהָב טָהוֹר עָלָיו, אֲשֶׁר מַה שֶּׁנֶּאֱמַר בְּדִבְרֵי חֲזַ"ל אַשְׁרֵי מִי שֶׁבָּא לְכָאן וְתַלְמוּדוֹ בְיָדוֹ – נֶאֱמַר עַתָּה עַל סִפְרֵי רַבֵּנוּ הַקָּדוֹשׁ זְצוּקַ"ל. 
כִּי רַבֵּנוּ זַ"ל אָמַר, שֶׁבְּכָל תּוֹרָה וְתּוֹרָה כָּלוּל כָּל דִּבְרֵי תּוֹרָה שֶׁבִּכְתָב וְשֶׁבְּעַל פֶּה, וְהֵם שָׁרְשֵׁי הַתּוֹרָה. אָמַרְתִּי בְלִבִּי לְפִי קָט שִׂכְלִי: זֶה אֶצְלִי בִּלְתִּי אֶפְשָׁר – כִּי הֵם עֲמוּקִים מְאֹד מִנִּי יָם, וּבְכָל תּוֹרָה וְתוֹרָה נִמְצָא כְּלָלִים וּפְרָטִים וְעִנְיָנִים רַבִּים וְסוֹדוֹת גְּדוֹלוֹת. 
עַל כֵּן יִשַּׁבְתִּי בְדַעְתִּי: אִם אֵינֶנִּי יָכוֹל לַעֲשׂוֹת כֹּלָּא מִצְוָה – עַל כָּל פָּנִים אַעֲבִיד חֵלֶק מֵהַמִּצְוָה, וְאַצִּיגָה כָּל הַצֵּרוּפֵי אוֹתִיּוֹת הַנִּמְצָאִים בִּסְפָרָיו הַקְּדוֹשִׁים – וְאֶעֱמֹד אוֹתָם עַל פִּי אָלֶף בֵּית, וְלַחֲזֹר מֵאָה וְאֶחָד פְּעָמִים – בִּכְדֵי לִהְיוֹת לִי לְזִכָּרוֹן בְּעָלְמָא דֵין וּבְעָלְמָא דְאָתֵי וּלְנֶצַח נְצָחִים.
וּבִפְרָט שֶׁשָּׁמַעְתִּי מֵאַדְמוֹ"ר מוֹהֲרַנַ"תְּ זְצוּקַ"ל, זְכוּתוֹ יָגֵן עָלֵינוּ אָמֵן, שֶׁסִּפֵּר לָנוּ שֶׁאֶחָד מֵאַנְשֵׁי שְׁלוֹמֵנוּ מִמְּקֹרָבָיו שָׁאַל אוֹתוֹ: אֵיךְ אֶזְכֶּה לְאַחַר פְּטִירָתִי לִשְׁאוֹל אֶת עַצְמִי לְרַבֵּנוּ זַ"ל, וְהֵשִׁיב לוֹ בְּזֶה הַלָּשׁוֹן: "הַאַתָּה אֵינְךָ יוֹדֵעַ אֵיךְ לִשְׁאוֹל עַצְמְךָ לְרַבֵּנוּ זַ"ל?! הֲלֹא אַתָּה זוֹכֵר הַצֵּרוּפֵי אוֹתִיּוֹת מִסִּפְרֵי רַבֵּנוּ זַ"ל"! וְהִזְכִּיר לוֹ אָז כַּמָּה צֵרוּפֵי אוֹתִיּוֹת. 
וְיַעֲמֹד לִי זְכוּת צֵרוּפָיו וּרְמָזָיו וְסוֹדוֹתָיו הַנִּפְלָאִים וְהַנּוֹרָאִים – לִי וְלְזַרְעִי וְזֶרַע זַרְעִי עַד סוֹף כָּל הַדּוֹרוֹת, שֶׁנִּהְיֶה דְּבֵקִים תָּמִיד בְּתוֹרַת ה' אֲשֶׁר מָסַר לָנוּ עַל יְדֵי מֹשֶׁה נְבִיאוֹ – נֶאֱמַן בֵּיתוֹ, וְעַל יְדֵי כָּל הַצַּדִּיקִים הָאֲמִתִּיִּים נֶאֶמְנֵי בֵּית ה', וְשֶׁיִּהְיֶה תִּקּוּן לְנַפְשׁוֹתֵינוּ וְרוּחוֹתֵינוּ וְנִשְׁמוֹתֵינוּ לְעוֹלְמֵי עַד וּלְנֶצַח נְצָחִים. וְאִמְרוֹת ה' צְרוּפָה – מָגֵן הוּא לְכֹל הַחֹסִים בּוֹ, צְרוּפָה אִמְרָתְךָ מְאֹד וְעַבְדְּךָ אֲהֵבָהּ. 
נֹעַם תִּפְאֶרֶת נַחֲלָתֵנוּ, בְּצִלּוֹ נִחְיֶה, יִשְׁמְעוּ הַמִּתְהַלְלִים וְיֹאמְרוּ דְּבָרָיו הַשְׁמִיעֵנו (ר"ת נתן בן יהודה).

שָׁלֵ"ו רָאשֵׁי תֵבוֹת וְתוֹרָתְךָ לֹא שָׁכָחְתִּי. [ליקו"מ מ"ו]
שָׁלֵ"ו בְּגִימַטְרִיָּא שְׁלֹשָׁה פְעָמִים הֲוַיָ"ה וּשְׁלֹשָׁה פְעָמִים אֱלֹהִים. [שם]
שָׁלֵ"ו רָאשֵׁי תֵבוֹת וְיָשֵׂם לְךָ שָׁלוֹם. [שָׁם]
שָׂפָה רָאשֵׁי תֵבוֹת שָׂכָר פְּרִי הַבָּטֶן. [ליקו"מ מ"ח]
שְׂאֵת רָאשֵׁי תֵבוֹת שְׂפַת אֱמֶת תִּכּוֹן. [שם]
שַׁבָּת רָאשֵׁי תֵבוֹת בַּיִת תְּחַלֵּק שָׁלָל. [ליקו"מ מ"ט]
שָׁלוֹם רָאשֵׁי תֵבוֹת וְדַע מַה שֶּׁתָּשִׁיב לְאֶפִּיקוֹרֶס. [ליקו"מ ס"ב]
שְׁתִיָּה רָאשֵׁי תֵבוֹת הֶן תָּוִי שַׁדַּי יַעֲנֵנִי. [ליקו"מ ס"א]
שָׁלוֹם רָאשֵׁי תֵבוֹת וּמַגִּיד לְאָדָם מַה שֵּׂחוֹ. [ליקו"מ רל"ט]
שַׁבָּת שִׁין בַּת. [ליקו"מ ח"ב ס"ז]
שִׁידּוּך רָאשֵׁי תֵבוֹת שִׂפְתֵי כֹהֵן יִשְׁמְרוּ דַעַת וְתוֹרָה. [ליקו"מ ח"ב פ"ט]
שִׁמְעוֹן אוֹתִיּוֹת עָוֹן מָשׁ. [שיחות הר"ן מ"ד] וְשָׁם מְבֹאָר גֹּדֶל מַעֲלַת הַצֵּרוּפֵי אוֹתִיּוֹת. 
שַׁדְכָן רָאשֵׁי תֵבוֹת כָּרְסְיֵהּ שְׁבִבִין דִּי נוּר. [חיי"מ סימן פ"ד]
שׁוֹפָר רָאשֵׁי תֵבוֹת שְׂחוֹק פִּינוּ וּלְשׁוֹנֵנוּ רִנָּה. [חיי"מ סימן צ"ו]
שֻׁלְחָן בְּגִימַטְרִיָּא קְלִיפָּה נִדְּחָה פִּיהָ עִם הַכּוֹלֵל. [חיי"מ סימן תע"ו] 
שֻׁלְחָן בְּגִימַטְרִיָּא הֲוַיָ"ה יְקָרְבֶנּוּ. [שם]
שֻׁלְחָן בְּגִימַטְרִיָּא צַו רְפוּאָה. וּבְגִימַטְרִיָּא שַׁ"ךְ חֶסֶ"ד. וְגִימַטְרִיָּא חַשְׁמַ"ל הַמַּלְבִּישׁ עֲשָׂרָה סְפִירוֹת. [שם]
שֻׁלְחָן גִּימַטְרִיָּא ש"ע נְהוֹרִין עִם חַ"י, בְּחִינַת בְּאוֹר פְּנֵי מֶלֶךְ חַיִּים. [שם] 
שָׁלוֹם רָאשֵׁי תֵבוֹת מֵאָשֵׁר שְׁמֵנָה לַחְמוֹ וְהוּא. [לקוטי הלכות הל' בית הכנסת ו, נח.]

תּוֹרָה הִיא טַנְתָּ"א. טְעָמִים נְקֻדּוֹת תָּגִין אוֹתִיּוֹת. [ליקו"מ י"ג]
תֶּשִׁי רָאשֵׁי תֵבוֹת תִּכָּלֵם שִׁבְעַת יָמִים. [ליקו"מ כ"א]
תִּשְׁרֵי גִּימַטְרִיָּא שְׁנֵי פְעָמִים חוֹתָם עִם שְׁנֵי תֵבוֹת. [ליקו"מ כ"ב]
תּוֹלָע בְּחִינַת עוֹלַת. [ליקו"מ כ"ח]
תֵּבֵל גִּימַטְרִיָּא שְׁנֵי פְעָמִים רי"ו, בְּחִינַת מְחָאַת כַּפָּיִם. [ליקו"מ מ"ו]
תִּשְׁרֵי לְשׁוֹן שִׁירָה. [ליקו"מ מ"ט]
תִּקְעוּ גִּימַטְרִיָּא שְׁנֵי פְעָמִים ע"ב רי"ו, בְּחִינַת מְחָאַת כַּפָּיִם. [שם]
תְּהִלִּים גִּימַטְרִיָּא לִ-י-לִ-י-ת עִם חֲמִשָּׁה אוֹתִיּוֹת שֶׁל שְׁמָהּ. [ליקו"מ ר"ה]
תְּהִלִּים גִּימַטְרִיָּא שְׁנֵי שֵׁמוֹת אֵ"ל אֱלֹהִים בְּמִלּוּאָם. [שם]
תַּלְמוּד גִּימַטְרִיָּא אוֹתִיּוֹת לִ-י-לִ-י-ת, לְהַכְנִיעָהּ. [ליקו"מ רי"ד]
תִּשְׁרֵי רָאשֵׁי תֵבוֹת יָם שִׁבַּרְתָּ רָאשֵׁי תַנִּינִים. [ליקו"מ ח"ב ח']
תִּשְׁרֵי סוֹפֵי תֵבוֹת וְאַתֶּם יְדַעְתֶּם אֶת נֶפֶשׁ הַגֵּר כִּי. [שם]
תְּרוּמָה סוֹפֵי תֵבוֹת לַחֲזוֹת בְּנֹעַם הֲוַיָ"ה וּלְבַקֵּר בְּהֵיכָלוֹ [ליקו"מ ח"ב ע"א]
תְּהִלִּים תְּשׁוּבָה סוֹפֵי תֵבוֹת וְאֵלֶּה שְׁמוֹת בְּנֵי יִשְׂרָאֵל הַבָּאִים מִצְרָיְמָה אֵת יַעֲקֹב אִישׁ וּבֵיתוֹ. [ליקו"מ ח"ב ע"ג]
תְּקִיעָה תְּרוּעָה תְּקִיעָה רָאשֵׁי תֵבוֹת תַּאֲוַת רְשָׁעִים תֹּאבֵד. [ליקו"מ ח"ב פ"ח]
תְּהִלִּם רָאשֵׁי תֵבוֹת לִמְטַר הַשָּׁמָיִם תִּשְׁתֶּה מָיִם. [ספר המידות סגולה, חלק שני, ד.]
תָּמִד גִּימַטְרִיָּא אַרְבָּעָה פְעָמִים אֶלֶף, תִּקּוּן שְׁלֹשֶׁת אֲלָפִים מָשָׁל.
